# Factory hard blocks (class D): what no agent of a factory run may do.
#
# One file, three readers: the hooks module (register.js, tool.call), the
# PreToolUse guard (factory-done-guard.sh via factory-hardblock.sh) and
# factory-deny.sh, which copies the `deny` lines into a project's
# .claude/settings.json. Change a rule here and every layer changes with it.
#
# Fields are separated by " ## ". Regexes are read by Python
# (factory-hardblock.py) and by JavaScript (register.js), with [:space:] and
# [:alnum:] spelled out for both; keep to what the two read alike (lookahead
# is fine, no \b or \d, no lookbehind).
#
# A rule that refuses a command halts the run (.factory/STOP). What a command
# only prints or searches for is not read as a command: the quoted text of a
# command that runs nothing (factory-block T1 'copy the key into .env',
# grep -rn "git push" scripts), a heredoc cat or tee writes to a file, a
# comment.
#
#   hard_blocks_version: N        raised whenever a rule is added or loosened;
#                                 an autonomy consent written under another
#                                 number counts for nothing (factory-autonomy)
#   bash ## name ## regex          a Bash command (newlines are read as ';')
#   path ## name ## any|root ## X [## create-only]
#                                  a file written by a write tool, or by a Bash
#                                  or PowerShell redirect / tee / cp / mv / rm /
#                                  sed -i / Set-Content ..., or by inline code
#                                  (python -c, node -e, a heredoc) that names it
#                                  and calls a write; "root" is relative to the
#                                  project root, "any" matches at any depth;
#                                  create-only lets rm through (removing the file
#                                  only takes a green away). The folders such a
#                                  path starts with (.factory, gates, .git) are
#                                  held too: writing into one, moving or removing
#                                  it (rule name protected-dir).
#   deny ## rule                   a Claude Code permissions.deny rule
#   group ## name ## question      an optional ban: the developer may allow it per
#                                  project (.factory/config.json,
#                                  hard_blocks.allow: ["name", ...]), asked once by
#                                  /factory:init and changed with /factory:bans
#   member ## group ## bash|path|deny ## X
#                                  what a group lifts: the name of a bash or path
#                                  rule, or a deny rule line. An allowed group is
#                                  dropped from all three layers at once.
#
# Matching ignores letter case and reads every spelling of a path as one: "\"
# as "/", the project root in any form (C:\p, /c/p, $PWD, $(pwd), ~+,
# $CLAUDE_PROJECT_DIR, ...) removed, "dir/../" collapsed. Inline code that
# writes and names a protected path is refused without halting the run: that
# match is a guess.

hard_blocks_version: 3

# no agent may write, and it voids an autonomy consent given before it.
