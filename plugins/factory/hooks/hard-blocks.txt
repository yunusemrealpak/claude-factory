# Factory hard blocks (class D): what no agent of a factory run may do.
#
# One file, three readers: the hooks module (register.js, tool.call), the
# PreToolUse guard (factory-done-guard.sh via factory-hardblock.sh) and
# factory-deny.sh, which copies the `deny` lines into a project's
# .claude/settings.json. Change a rule here and every layer changes with it.
#
# Fields are separated by " ## ". Regexes are POSIX ERE that also parse as
# JavaScript once [[:space:]] / [:alnum:] are spelled out, which register.js
# does; keep to that subset (no \b, \d, lookahead).
#
#   hard_blocks_version: N        raised whenever a rule is added or loosened;
#                                 an autonomy consent written under another
#                                 number counts for nothing (factory-autonomy)
#   bash ## name ## regex          a Bash command (newlines are read as ';')
#   path ## name ## any|root ## X  a file written by a write tool, or by a Bash
#                                  redirect / tee / cp / mv / rm / sed -i ...;
#                                  "root" is relative to the project root,
#                                  "any" matches at any depth
#   deny ## rule                   a Claude Code permissions.deny rule
#   group ## name ## question      an optional ban: the developer may allow it per
#                                  project (.factory/config.json,
#                                  hard_blocks.allow: ["name", ...]), asked once by
#                                  /factory:init and changed with /factory:bans
#   member ## group ## bash|path|deny ## X
#                                  what a group lifts: the name of a bash or path
#                                  rule, or a deny rule line. An allowed group is
#                                  dropped from all three layers at once.

hard_blocks_version: 1

bash ## git-force ## (^|[^[:alnum:]_.-])git[[:space:]]+[^;&|]*[[:space:]]--force
bash ## git-tag ## (^|[^[:alnum:]_.-])git(([[:space:]]+(-C|-c|--git-dir|--work-tree|--namespace)[[:space:]]+[^[:space:]]+)|([[:space:]]+--?[A-Za-z][A-Za-z-]*(=[^[:space:]]*)?))*[[:space:]]+tag([[:space:];&|)]|$)
bash ## git-no-verify ## (^|[^[:alnum:]_.-])git[[:space:]]+[^;&|]*[[:space:]]--no-verify([[:space:];&|)]|$)

bash ## gh-release ## (^|[^[:alnum:]_.-])gh[[:space:]]+([^;&|]*[[:space:]])?release([[:space:];&|)]|$)

# --- optional bans: the developer decides per project ------------------------
# Everything above is a ban nobody may lift. These are bans a developer may not
# want: release work they would rather the factory (or their own session) did.
# Allowing one is written by the developer's own session into config.json, which
# no agent may write, and it voids an autonomy consent given before it.
group ## store-deploy ## Store and TestFlight uploads (fastlane, altool, notarytool, codemagic): may agents and this session run them?
member ## store-deploy ## bash ## store-deploy
member ## store-deploy ## deny ## Bash(fastlane *)
member ## store-deploy ## deny ## Bash(*altool *)
member ## store-deploy ## deny ## Bash(*notarytool *)
member ## store-deploy ## deny ## Bash(codemagic*)
